int ldrPin = A0;
int ledPin = 13;

int value;

void setup() {
  pinMode(ledPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  value = analogRead(ldrPin);

  Serial.println(value);

  if (value < 500) {
    digitalWrite(ledPin, HIGH);
  }
  else {
    digitalWrite(ledPin, LOW);
  }

  delay(200);
}
